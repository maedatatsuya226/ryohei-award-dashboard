    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (!sheet) sheet = ss.insertSheet(CONFIG.DATA_SHEET);

      // エントリー用
      '推薦理由', '特筆エピソード', 
      '評価1_医療貢献(済)', '評価1_備考', '評価2_業務改善(済)', '評価2_備考', '評価3_対外活動(済)', '評価3_備考', 
      '評価4_多職種連携(済)', '評価4_備考', '評価5_次世代(済)', '評価5_備考', '評価6_その他(済)', '評価6_備考',

      '活動テーマ', '活動開始年', '活動概要', '活動の成果', '継続性・発展性', '波及効果・拡張性', '社会貢献度', '自己分析・改善点',
      // 共通下部
      '備考', '推薦者_所属', '推薦者_役職', '推薦者_氏名', '添付ファイルURL',
      // 事務局判定用（手動入力）
      '2次審査進出', '最終選考進出'

    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e0e0e0');
    sheet.setFrozenRows(1);
    
    // 2. 「設定」シートの自動生成
    let settingSheet = ss.getSheetByName('設定');
    if (!settingSheet) {
      settingSheet = ss.insertSheet('設定');
      settingSheet.getRange('A1:B1').setValues([['設定項目', '設定値']]).setFontWeight('bold').setBackground('#f3f4f6');

// 4. テンプレートへの流し込み＆PDF出力処理（変更なし）
// ==========================================
function generatePDF(docType, data) {
  let newSheet = null; 
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  try {
    const sheetName = docType === 'entry' ? CONFIG.ENTRY_TEMPLATE : CONFIG.FINAL_TEMPLATE;
    const templateSheet = ss.getSheetByName(sheetName);
    const folder = DriveApp.getFolderById(CONFIG.OUTPUT_FOLDER_ID);
    if (!templateSheet) throw new Error(`シート「${sheetName}」が見つかりません。`);
    
    const docTypeName = docType === 'entry' ? 'エントリー' : '最終選考';
    const timestamp = Utilities.formatDate(new Date(), "JST", "MMdd_HHmm");
    const newSheetName = `【${docTypeName}】${data.candidateName}_${timestamp}`;
    
    newSheet = templateSheet.copyTo(ss).setName(newSheetName);
    const cellMap = docType === 'entry' ? CELL_MAP_ENTRY : CELL_MAP_FINAL;

    const keys = Object.keys(cellMap);
    keys.forEach(key => {
      const cellAddress = cellMap[key];
      if (cellAddress && data[key] !== undefined) {
        const cell = newSheet.getRange(cellAddress);
        
        if (key.includes('_check')) {
          const rule = cell.getDataValidation();
          if (rule !== null && rule.getCriteriaType() === SpreadsheetApp.DataValidationCriteria.CHECKBOX) {
            cell.setValue(data[key] ? true : false);
          } else {
            let text = cell.getValue().toString();
            if (text.length > 3) {
              const cleanText = text.replace(/^[□☐☑■✔\s]+/, '');
              cell.setValue((data[key] ? '☑ ' : '□ ') + cleanText);
            } else {
              cell.setValue(data[key] ? '☑' : '□');
            }
          }
        } else {
          // テキストの流し込み
          const textValue = String(data[key] || "");
          cell.setValue(textValue);
          
          // ▼▼ 自動文字縮小ロジック ▼▼
          // 文字列の長さによって段階的にフォントサイズを小さくする
          const length = textValue.length;
          let fontSize = 10; // デフォルト（基準）サイズ
          
          // 項目の性質（通常1行か、長文エリアか）で基準を変えたい場合は
          // セルの元のフォントサイズを取得して基準にするのが安全です
          const originalFontSize = cell.getFontSize() || 10;
          
          if (length > 200) {
            fontSize = Math.max(originalFontSize - 4, 6); // 最小6pt
          } else if (length > 150) {
            fontSize = Math.max(originalFontSize - 3, 7);
          } else if (length > 100) {
            fontSize = Math.max(originalFontSize - 2, 8);
          } else if (length > 60) {
            fontSize = Math.max(originalFontSize - 1, 9);
          } else {
            fontSize = originalFontSize;
          }
          
          cell.setFontSize(fontSize);
          // ▲▲ 自動文字縮小ロジックここまで ▲▲
        }
      }
    });
    SpreadsheetApp.flush();

    const url = `https://docs.google.com/spreadsheets/d/${ss.getId()}/export?exportFormat=pdf&format=pdf&size=A4&portrait=true&fitw=true&sheetnames=false&printtitle=false&pagenumbers=false&gridlines=false&fzr=false&gid=${newSheet.getSheetId()}`;
    const options = { headers: { 'Authorization': 'Bearer ' + ScriptApp.getOAuthToken() }, muteHttpExceptions: true };
    const response = UrlFetchApp.fetch(url, options);
    if (response.getResponseCode() !== 200) throw new Error('PDF生成失敗');

    folder.createFile(response.getBlob().setName(`【${docTypeName}】良平アワード_${data.candidateName}.pdf`));
  } catch (e) {
    if (newSheet && e.message.includes('PDF生成失敗')) ss.deleteSheet(newSheet); 
    throw e;
  }
}

// 5. フロントエンドからのデータ受信API（★全35項目蓄積に修正）
// ==========================================
function doPost(e) {
  try {
    const params = JSON.parse(e.postData.contents);
    
    // ▼▼ 評価データの保存処理分岐 ▼▼
    if (params.action === 'evaluate') {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      
      // 1次審査の場合は別シートに保存
      if (params.phase === '1次審査') {
        let eval1Sheet = ss.getSheetByName("1次審査結果");
        if (!eval1Sheet) {
          eval1Sheet = ss.insertSheet("1次審査結果");
          eval1Sheet.appendRow(["タイムスタンプ", "対象者", "所属", "職種", "評価者名", "スコア(★1-5)"]);
          eval1Sheet.getRange("A1:F1").setFontWeight("bold").setBackground("#fef08a");
          eval1Sheet.setFrozenRows(1);
        }
        eval1Sheet.appendRow([
          new Date(),
          params.candidateName || '',
          `${params.hospitalName || ''} / ${params.department || ''}`,
          params.jobTitle || '',
          params.reviewerName || '',
          params.score || ''
        ]);
        return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: '1次審査の評価を保存しました' })).setMimeType(ContentService.MimeType.JSON);
      } else {
        // 2次・最終審査用（既存処理）
        let evalSheet = ss.getSheetByName("評価結果");
        if (!evalSheet) {
          evalSheet = ss.insertSheet("評価結果");
          evalSheet.appendRow(["タイムスタンプ", "対象者", "所属", "職種", "フェーズ", "評価者名", "スコア", "コメント"]);
          evalSheet.getRange("A1:H1").setFontWeight("bold").setBackground("#f3f4f6");
          evalSheet.setFrozenRows(1);
        }
        evalSheet.appendRow([
          new Date(),
          params.candidateName || '',
          `${params.hospitalName || ''} / ${params.department || ''}`,
          params.jobTitle || '',
          params.phase || '2次審査',
          params.reviewerName || '',
          params.score || '',
          params.comment || ''
        ]);
        return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: '評価を保存しました' })).setMimeType(ContentService.MimeType.JSON);
      }
    }
    // ▲▲ 評価データの保存処理分岐 ここまで ▲▲

    const docType = params.docType;
    let fileUrl = ""; 
    
    if (params.files && params.files.length > 0) {
      try {
        const folder = DriveApp.getFolderById(CONFIG.OUTPUT_FOLDER_ID);
        const urls = [];
        for (let i = 0; i < params.files.length; i++) {
          const f = params.files[i];
          let base64Data = f.fileData;
          if (base64Data.includes(',')) base64Data = base64Data.split(',')[1];
          const decodedFile = Utilities.base64Decode(base64Data);
          const blob = Utilities.newBlob(decodedFile, f.mimeType || 'application/octet-stream', `【添付】${params.candidateName}_${f.fileName}`);
          urls.push(folder.createFile(blob).getUrl());
        }
        // 複数URLを改行で結合してスプレッドシートの1セルに保存
        fileUrl = urls.join('\n'); 
      } catch (err) {
        console.error("添付ファイルの保存に失敗: " + err.stack);
      }
    } else if (params.fileData && params.fileName) {
      // 古いバージョンの1ファイル処理（互換性用）
      try {
        const folder = DriveApp.getFolderById(CONFIG.OUTPUT_FOLDER_ID);
        let base64Data = params.fileData;
        if (base64Data.includes(',')) base64Data = base64Data.split(',')[1];
        const decodedFile = Utilities.base64Decode(base64Data);
        const blob = Utilities.newBlob(decodedFile, params.mimeType || 'application/octet-stream', `【添付】${params.candidateName}_${params.fileName}`);
        fileUrl = folder.createFile(blob).getUrl(); 
      } catch (err) {
        console.error("添付ファイルの保存に失敗: " + err.stack);
      }
    }

    if (docType) {
      generatePDF(docType, params);
    } else {
      throw new Error("docTypeが指定されていません。");
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);
    
    // ヘッダー順に完全対応させた35項目のデータ
    const newRow = [
      new Date(), 
      docType === 'entry' ? 'エントリー' : '最終選考', 
      params.hospitalName || '', params.candidateName || '', params.age || '', params.yearsOfService || '', params.department || '', params.jobTitle || '',
      
      // エントリー用（チェックボックスは☑/□の文字に変換してシートに記録）
      params.reason || '', params.episode || '',
      params.eval1_check ? '☑' : '□', params.eval1_note || '',
      params.eval2_check ? '☑' : '□', params.eval2_note || '',
      params.eval3_check ? '☑' : '□', params.eval3_note || '',
      params.eval4_check ? '☑' : '□', params.eval4_note || '',
      params.eval5_check ? '☑' : '□', params.eval5_note || '',
      params.eval6_check ? '☑' : '□', params.eval6_note || '',
      
      // 最終選考用
      params.theme || '', params.startYear || '', params.summary || '', params.results || '', params.continuity || '', params.rippleEffect || '', params.socialContribution || '', params.selfAnalysis || '',
      
      // 共通下部
      params.remarks || '', params.recommenderDept || '', params.recommenderTitle || '', params.recommenderName || '', fileUrl,
      
      // 事務局判定用フラグ枠（初期値は空）
      false, false
    ];
    dataSheet.appendRow(newRow);

    return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: '送信が完了し、PDFが生成されました' })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

// 6. 審査員ダッシュボード用のデータ送信API
// ==========================================
function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // ▼▼ 評価者一覧の取得用API ▼▼
    if (e && e.parameter && e.parameter.action === 'getReviewers') {
      const reviewerSheet = ss.getSheetByName("審査者一覧");
      if (!reviewerSheet) {
        // 万が一シートがない場合は空配列を返す
        return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
      }
      
      const reviewerData = reviewerSheet.getDataRange().getValues();
      let reviewers = [];
      if (reviewerData.length > 1) {
        // A列（0番目）の2行目以降を取得（空白は除外）
        reviewers = reviewerData.slice(1).map(row => row[0]).filter(name => name !== '');
      }
      
      return ContentService.createTextOutput(JSON.stringify(reviewers)).setMimeType(ContentService.MimeType.JSON);
    }
    // ▲▲ 評価者一覧の取得用API ここまで ▲▲

    // ▼▼ 現在のフェーズ取得 ▼▼
    let currentPhase = '1次審査'; // デフォルト
    const settingSheet = ss.getSheetByName('設定');
    if (settingSheet) {
      const phaseVal = settingSheet.getRange('B2').getValue();
      if (phaseVal) currentPhase = phaseVal;
    }

    const sheet = ss.getSheetByName(CONFIG.DATA_SHEET);
    if (!sheet) throw new Error("データシートが見つかりません。");

    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);
    
    const result = rows.filter(row => row[0] !== "").map((row, index) => {
      let obj = { id: index + 1 }; 
      headers.forEach((header, colIndex) => {
        if (row[colIndex] instanceof Date) {
          obj[header] = Utilities.formatDate(row[colIndex], "JST", "yyyy/MM/dd HH:mm");
        } else {
          obj[header] = row[colIndex];
        }
      });
      return obj;
    });

    // ▼▼ 評価済みデータの紐付け処理 ▼▼
    // 1次審査の場合は「1次審査結果」、2次/最終の場合は「評価結果」シートから取得
    const evalSheetName = currentPhase === '1次審査' ? '1次審査結果' : '評価結果';
    const evalSheet = ss.getSheetByName(evalSheetName);
    
    if (evalSheet) {
      const evalData = evalSheet.getDataRange().getValues();
      if (evalData.length > 1) {
        const evalHeaders = evalData[0];
        const targetIdx = evalHeaders.indexOf("対象者");
        const reviewerIdx = evalHeaders.indexOf("評価者名");
        const deptIdx = evalHeaders.indexOf("所属"); // 「病院名 / 部署」形式で保存されている
        const phaseIdx = evalHeaders.indexOf("フェーズ"); // 2次/最終の場合

        if (targetIdx !== -1 && reviewerIdx !== -1) {
          // 同名の対象者（例: 部署単位のエントリー）を区別するため、
          // 「対象者名｜所属」を複合キーにして紐付ける
          const makeKey = (name, affiliation) => `${String(name).trim()}|${String(affiliation || '').trim()}`;

          const evalMap = {};
          for (let i = 1; i < evalData.length; i++) {
             const targetName = evalData[i][targetIdx];
             const reviewerName = evalData[i][reviewerIdx];
             const affiliation = deptIdx !== -1 ? evalData[i][deptIdx] : '';
             // シートにフェーズ列がある場合、現在のフェーズと一致するか確認
             let isValidPhase = true;
             if (phaseIdx !== -1 && currentPhase !== '1次審査') {
               isValidPhase = (evalData[i][phaseIdx] === currentPhase);
             }

             if (targetName && reviewerName && isValidPhase) {
                const key = makeKey(targetName, affiliation);
                if (!evalMap[key]) evalMap[key] = [];
                if (!evalMap[key].includes(reviewerName)) {
                   evalMap[key].push(reviewerName);
                }
             }
          }

          result.forEach(candidate => {
            // 評価結果シートの「所属」列と同じ形式（病院名 / 部署）でキーを組み立てる
            const affiliation = deptIdx !== -1
              ? `${candidate["病院名"] || ''} / ${candidate["所属部署"] || ''}`
              : '';
            candidate.evaluatedBy = evalMap[makeKey(candidate["対象者名"], affiliation)] || [];
          });
        }
      }
    }
    // ▲▲ 評価済みデータの紐付け処理 ここまで ▲▲

    return ContentService.createTextOutput(JSON.stringify({ 
      status: 'success', 
      phase: currentPhase,
      data: result 
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}
